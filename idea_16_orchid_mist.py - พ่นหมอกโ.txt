class Button:
    def __init__(self, index):
        self.index, self.down, self.clicked = index, False, False

    def sample(self):
        try:
            now_down = buttons.pressed(self.index)
        except Exception:               # index ผิด/อ่านปุ่มไม่ได้ ต้องไม่ทำให้ทั้งโปรแกรมล้ม
            now_down = False
        if now_down and not self.down:
            self.clicked = True
        self.down = now_down

    def pressed_now(self):
        fired, self.clicked = self.clicked, False
        return fired


def wait_ms(ms, btns):
    t0 = time.ticks_ms()
    while time.ticks_diff(time.ticks_ms(), t0) < ms:
        for b in btns:
            b.sample()
        time.sleep_ms(20)


def matrix_countdown(sec_left, shown):
    if shown <= 0 or sec_left == 0:
        rgbmatrix.scroll("")
        rgbmatrix.clear()
    if sec_left:
        rgbmatrix.score(sec_left, rgbmatrix.CYAN)
    return sec_left

def pump_seconds(sec):
    # ตรวจเวลาที่สั่ง: ไม่ใช่จำนวนเต็มบวก = ค่าตั้งต้น  เกินเพดาน = เพดาน
    if isinstance(sec, bool) or not isinstance(sec, int) or sec <= 0:  # bool ไม่ใช่จำนวนเต็ม (สัญญาข้อ 4)
        sec = PUMP_DEFAULT_S
    return min(sec, PUMP_MAX_S)


def ascii_only(text):
    return "".join(c for c in str(text) if " " <= c <= "~")[:20]


# ช่วงความชื้น: 0 = แห้ง, 1 = พอดี, 2 = ชื้นเกิน -> (คำบนจอ, สี)

def band_id(rh, prev):
    # มี hysteresis: ถ้าเพิ่งอยู่นอกช่วง ต้องกลับเข้ามาลึกกว่าเส้นอีก RH_HYST ถึงนับว่า "พอดี"
    if prev == 0 and rh < RH_DRY + RH_HYST:
        return 0
    if prev == 2 and rh > RH_WET - RH_HYST:
        return 2
    if rh < RH_DRY:
        return 0
    if rh > RH_WET:
        return 2
    return 1


_temp_err = ""                         # เหตุผลที่อ่าน SHT40 ไม่ได้ (ว่าง = อ่านได้) แสดงบนจอและส่งขึ้นเว็บ


def read_temp():
    # คืน (อุณหภูมิ °C, แหล่ง) แหล่ง = "SHT40" (จริง) หรือ "VR2" (จำลอง)
    global _temp_err
    if USE_REAL_TEMP:
        if sensors is None:
            _temp_err = "no sensors module"
        else:
            try:
                t = sensors.sht40.temperature() + TEMP_OFFSET
                if t == t and TEMP_VALID_MIN <= t <= TEMP_VALID_MAX:     # t == t กัน NaN
                    _temp_err = ""
                    return t, "SHT40"
                _temp_err = "out of range"
            except Exception as e:
                _temp_err = str(e)[:20] or "read fail"
    return TEMP_MIN_C + knob_percent(TEMP_POT) * (TEMP_MAX_C - TEMP_MIN_C) / 100, "VR2"


def calc_vpd(t, rh):
    # VPD (kPa) = ความดันไอน้ำอิ่มตัว x (1 - RH/100)
    svp = 0.6108 * math.exp(17.27 * t / (t + 237.3))
    return svp * (1 - rh / 100)


# ความเครียดของพืช: 0 = ไม่เครียด, 1 = เริ่มเครียด, 2 = เครียด -> (คำบนจอ, สี)
STRESS = (("ไม่เครียด", COL_PETAL), ("เริ่มเครียด", COL_DRY), ("เครียด!", COL_BAD))


def stress_id(vpd):
    if VPD_LOW <= vpd <= VPD_HIGH:
        return 0
    if VPD_LOW - VPD_MARGIN <= vpd <= VPD_HIGH + VPD_MARGIN:
        return 1
    return 2


def handle_command(raw, tank):
    # ตัดสินคำสั่งหนึ่งใบ -> (ทำอะไร, วินาทีหรือข้อความ, ข้อความขึ้นจอ, สี)  ไม่แตะของจริงในนี้
    try:
        cmd = json.loads(raw.decode())
    except ValueError:
        cmd = None
    if not isinstance(cmd, dict):      # 5, null, [] ก็เป็น JSON ได้ แต่ไม่ใช่คำสั่ง
        return "deny", 0, "อ่านคำสั่งไม่ได้", COL_BAD
    act, on = cmd.get("cmd", ""), cmd.get("on", 1)
    if act == "led" and cmd.get("n", 0) != 0:
        return "deny", 0, "ไม่มีหัวพ่นที่ " + str(cmd.get("n"))[:6], COL_DRY
    if act in ("led", "pump") and not on:
        return "off", 0, "สั่งหยุดพ่นหมอก", COL_PETAL
    if act in ("led", "pump") and tank < TANK_MIN:
        return "deny", 0, "ถังหมอกเหลือ %d%% ไม่พ่น" % tank, COL_BAD
    if act in ("led", "pump"):
        sec = pump_seconds(cmd.get("sec", PUMP_DEFAULT_S))
        return "on", sec, "พ่นหมอก %d วิ" % sec, COL_MIST
    if act == "auto":                  # เปิด/ปิดพ่นอัตโนมัติจากเว็บ (ไม่แตะปั๊มโดยตรง)
        return "auto", 1 if on else 0, "พ่นอัตโนมัติ: " + ("เปิด" if on else "ปิด"), COL_PINK
    if act == "beep":
        return "beep", 0, "เจ้าของสวนกล้วยไม้เรียก!", COL_PINK
    if act == "say":
        text = ascii_only(cmd.get("text", ""))
        return "say", text, "ข้อความ: " + (text or "ไม่มีอักษรอังกฤษ"), COL_PINK
    if act == "ack":  # ack เป็นของ sf2_04
        return "", 0, "ไม่มีแจ้งเตือน (sf2_04)", COL_DIM
    if act != "":
        return "deny", 0, "ไม่รู้จัก " + str(act)[:12], COL_DRY
    return "", 0, "", COL_DIM

def connect_broker(w):
    # broker สาธารณะไม่ตอบเป็นพัก ๆ: ลอง 3 ครั้ง client_id ใหม่ทุกครั้ง
    for n in (1, 2, 3):
        if n > 1:
            show_note(w, "ลองต่อ broker ใหม่ %d/3" % n, COL_DRY)
            ui.poll()
        try:
            if mqtt.connect(BROKER, port=1883, keepalive=60,
                            client_id=CLIENT_ID + "-%04x" % (time.ticks_ms() & 0xFFFF)):
                return True
        except OSError:
            pass
    return False


def connect_farm(w):
    # WiFi -> IP -> broker + subscribe ขั้นไหนพังบอกตรงนั้น
    show_note(w, "ต่อ WiFi... จอนิ่งได้", COL_DRY)
    ui.poll()                          # ป้ายต้องขึ้นจอก่อนบรรทัดที่บล็อก
    if not wifi.connect(WIFI_SSID, WIFI_PASS) or wifi.ip() == "0.0.0.0":
        return "ต่อ WiFi ไม่ได้"
    linked = connect_broker(w)         # ลองได้ 3 ครั้ง (ดู connect_broker)
    if not linked or not mqtt.subscribe(TOPIC_CMD):     # subscribe ต้องมาหลัง connect เสมอ
        return "broker ไม่ตอบ: รอ 1 นาทีแล้วรันใหม่"
    return ""


def send_report(n, rh, tank, running, bursts, rh_min, extra):
    # publish ตอนสายหลุดโยน OSError  (soil = ความชื้นอากาศ เพื่อให้สัญญาเดิมใช้ได้)
    try:
        body = {"id": TEAM, "n": n, "soil": rh, "tank": tank,
                "pump": 1 if running else 0, "sim": "soil tank temp_c",
                "bursts": bursts, "rh_min": rh_min}
        body.update(extra)             # temp_c vpd stress auto auto_wait auto_sec src dry wet
        mqtt.publish(TOPIC, json.dumps(body))
        return True
    except OSError:
        return False

    # True เฉพาะตอนค่าเปลี่ยนจากครั้งก่อน ใช้ข้ามการวาดจอที่ไม่จำเป็น (ลูปจะได้กลับมาสุ่มอ่านปุ่มเร็วขึ้น)
    if _last.get(key) == val:
        return False
    _last[key] = val
    return True


def show_link(w, ok):
    w["mq"].value(1 if ok else 0)
    w["mq_t"].text("MQTT: เชื่อมต่อแล้ว" if ok else "MQTT: ออฟไลน์")

def show_note(w, text, col):
    w["note"].color(col)
    w["note"].text(text)


def show_pump(w, sec_left):
    w["led"].value(1 if sec_left else 0)
    w["pump"].color(COL_MIST if sec_left else COL_DIM)
    w["pump"].text("กำลังพ่น อีก %d วิ" % sec_left if sec_left else "หยุด")
    w["left"].value(sec_left)


def show_air(w, rh, band, tank, rh_min, rh_max, manual):
    if not changed("air", (rh, band, tank, rh_min, rh_max, manual)):
        return
    word, col = BANDS[band]
    w["arc"].value(rh)
    w["arc"].color(col)
    w["rh"].text("%d %%" % rh)
    w["band"].color(col)
    w["band"].text(word if band == 1 else "! " + word)
    if manual:
        w["mode"].color(COL_DRY)
        w["mode"].text("โหมดมือ: VR3 = %d %%" % rh)
    else:
        w["mode"].color(COL_MIST)
        w["mode"].text("โหมด: อัตโนมัติ (VR1)")
    w["minmax"].text("ต่ำสุด %d  สูงสุด %d  (รอบนี้)" % (rh_min, rh_max))
    w["tank_bar"].value(tank)
    w["tank_bar"].color(COL_MIST if tank >= TANK_MIN else COL_BAD)
    w["tank"].text("%d %%" % tank)


def show_env(w, temp, temp_src, vpd, stress, auto_on):
    if not changed("env", (round(temp, 1), temp_src, round(vpd, 2), stress, auto_on)):
        return
    word, col = STRESS[stress]
    w["temp"].text("อุณหภูมิ %.1f C" % temp + ("" if temp_src == "SHT40" else " (จำลอง)"))
    w["vpd"].text("VPD %.2f kPa" % vpd)
    w["stress"].color(col)
    w["stress"].text("ความเครียด: " + word)
    w["auto"].text("พ่นอัตโนมัติ: " + ("เปิด" if auto_on else "ปิด"))


# ---- 6) โปรแกรมหลัก ----
class Stop(Exception):
    # ไม่ใช้ SystemExit: บอร์ดอาจค้าง
    pass


def stop(w, pump, msg, col=COL_BAD):
    # หยุดปั๊มหมอกก่อนเสมอ
    show_link(w, False)
    set_led(pump, False)
    rgbmatrix.scroll("")
    rgbmatrix.clear()
    show_note(w, msg, col)
    ui.poll()
    raise Stop


def act_on(w, raw, tank, now, pump_ms, pump_t0):
    do, arg, text, col = handle_command(raw, tank)
    if do == "off":
        pump_ms = 0
    elif do == "on":
        pump_ms, pump_t0 = arg * 1000, now
    elif do == "beep":
        beep("hit")
    elif do == "say" and arg and not pump_ms:
        rgbmatrix.scroll(arg, rgbmatrix.PURPLE, 80)
    tune = {"deny": "bad", "off": "stop", "on": "start", "say": "tap", "auto": "tap"}.get(do)
    if tune is not None:
        beep(tune)
    if text:
        show_note(w, text, col)
    return pump_ms, pump_t0, do, arg


def main():
    if hasattr(ui, "volume"):
        ui.volume(SPEAKER)
    w = build_screen()
    pump = led_named("RGB_BLUE")
    rgbmatrix.clear()
    if len(TEAM) != 6 or TEAM[:4] != "team" or not TEAM[4:].isdigit():
        stop(w, pump, "แก้ TEAM เป็นเลขกลุ่มก่อน")
    problem = connect_farm(w)
    if problem:
        stop(w, pump, problem)
    show_link(w, True)
    show_note(w, "ฟัง " + TOPIC_CMD, COL_PETAL)
    stop_btn = Button(SW_STOP_INDEX)
    auto_btn = Button(SW_AUTO_INDEX)
    BTNS.extend((stop_btn, auto_btn))
    got = n = pump_ms = pump_t0 = bursts = 0
    rh_min, rh_max = 100, 0            # ต่ำสุด/สูงสุดตั้งแต่เปิดเครื่อง
    shown = -1
    manual = False                     # True = ใช้ VR3 แทน VR1
    vr3_base = knob_percent(POT_MANUAL)  # ตำแหน่ง VR3 ตอนเริ่ม/ตอนกลับโหมดอัตโนมัติ
    band = 1
    auto_on = AUTO_DEFAULT             # พ่นอัตโนมัติเมื่อแห้ง
    auto_total = AUTO_SEC * 1000 + AUTO_WAIT_MS
    t_auto = time.ticks_add(time.ticks_ms(), -auto_total)   # รอบแรกพ่นได้ทันที
    temp, temp_src = read_temp()       # อุณหภูมิล่าสุด (อ่านใหม่ทุก TEMP_READ_MS)
    t_temp = time.ticks_ms()
    temp_warned = False
    t_alert = t_note = time.ticks_ms()
    alarm_until, alarm_i = None, 0     # None = ไม่มีเสียงเตือนดังอยู่
    send_now, last_pump = True, 0      # send_now = มีเหตุการณ์ใหม่ ส่งรายงานทันที (รอบแรกส่งเลย เว็บจะขึ้นเร็ว)
    face_band, face_dirty, face_warned = -1, False, False   # -1 = ยังไม่เคยวาดหน้า
    face_hold = time.ticks_ms()        # ก่อนเวลานี้ห้ามวาดทับ (ข้อความ say กำลังเลื่อน)
    t_send = t0 = time.ticks_ms()
    while time.ticks_diff(time.ticks_ms(), t0) < LISTEN_MS:
        now = time.ticks_ms()
        tick()
        st = (stop_btn.down, auto_btn.down)
        if changed("btn", st):          # ป้ายหัวจอ: กดค้างไว้แล้วต้องเห็น ON ถ้าไม่ขึ้นเลย แปลว่าเลขปุ่มไม่ตรงบอร์ด
            w["btn"].text("%s %s  %s %s" % (BTN_NAMES[0], "ON" if st[0] else "--", BTN_NAMES[1], "ON" if st[1] else "--"))
            w["btn"].color(COL_PINK if (st[0] or st[1]) else COL_DIM)
        if stop_btn.pressed_now():                                  # SW5 ไม่ผ่านเน็ตเลย
            if pump_ms:
                pump_ms = 0
                t_auto = now                                        # กันพ่นอัตโนมัติซ้ำทันทีหลังคนสั่งหยุด
                show_note(w, "หยุดพ่นฉุกเฉิน " + BTN_NAMES[0], COL_BAD)
                beep("stop")
            else:                       # กดแล้วตอบทุกครั้ง จะได้รู้ว่าบอร์ดรับปุ่มแล้ว
                show_note(w, BTN_NAMES[0] + ": ตอนนี้ปั๊มไม่ได้พ่นอยู่", COL_DIM)
                beep("tap")

        # โหมดมือ: VR3 ขยับ = สลับเข้า, SW6 = กลับอัตโนมัติ
        vr3 = knob_percent(POT_MANUAL)
        if auto_btn.pressed_now():
            if manual:
                manual, vr3_base, send_now = False, vr3, True
                show_note(w, "กลับโหมดอัตโนมัติ (VR1)", COL_MIST)
            else:
                show_note(w, BTN_NAMES[1] + ": อยู่โหมดอัตโนมัติอยู่แล้ว (หมุน VR3 เพื่อเข้าโหมดมือ)", COL_DIM)
            beep("tap")
        if not manual and abs(vr3 - vr3_base) >= MANUAL_MOVE:
            manual, send_now = True, True
            show_note(w, "โหมดมือ: ใช้ VR3 ตั้งความชื้น (กด %s กลับ)" % BTN_NAMES[1], COL_DRY)
            beep("manual")

        rh = vr3 if manual else knob_percent(POT_AUTO)
        tank = knob_percent(3)
        rh_min, rh_max = min(rh_min, rh), max(rh_max, rh)
        if time.ticks_diff(now, t_temp) >= TEMP_READ_MS:
            t_temp = now
            temp, temp_src = read_temp()
        if USE_REAL_TEMP and temp_src != "SHT40" and not temp_warned:     # บอกครั้งเดียวว่ากำลังใช้ค่าจำลอง
            temp_warned = True
            show_note(w, "อ่าน SHT40 ไม่ได้ (%s) ใช้ VR2 จำลองอุณหภูมิ" % (_temp_err or "?"), COL_DRY)
        elif temp_src == "SHT40":
            temp_warned = False
        vpd = calc_vpd(temp, rh)
        stress = stress_id(vpd)

        # เสียงเตือนเมื่อความชื้นอยู่นอก "พอดี": เตือนทันทีที่เปลี่ยนช่วง แล้วซ้ำทุก ALERT_EVERY_MS
        new_band = band_id(rh, band)
        if new_band != band:
            band = new_band
            t_alert = now
            send_now = True
            alarm_until = now + ALERT_SOUND_MS if band != 1 else None   # กลับมาพอดี = หยุดเสียงทันที
            alarm_i = 0
        elif band != 1 and time.ticks_diff(now, t_alert) >= ALERT_EVERY_MS:
            t_alert = now
            alarm_until, alarm_i = now + ALERT_SOUND_MS, 0
        if alarm_until is not None:
            if time.ticks_diff(alarm_until, now) <= 0:
                alarm_until = None
            elif time.ticks_diff(now, t_note) >= ALERT_NOTE_MS:
                t_note = now
                alarm_note("alert_dry" if band == 0 else "alert_wet", alarm_i)
                alarm_i += 1

        msg = mqtt.get_message()       # None = ยังไม่มีอะไรมา / (topic, bytes)
        if msg is not None:
            got += 1
            w["title"].text("ปั๊มหมอก (คำสั่ง %d)" % got)
            was_on = pump_ms
            pump_ms, pump_t0, did, darg = act_on(w, msg[1], tank, now, pump_ms, pump_t0)
            if did == "auto":           # เว็บเปิด/ปิดพ่นอัตโนมัติ
                auto_on, send_now = bool(darg), True
            if did == "say":            # ให้ข้อความเลื่อนบนจอ 8x8 ก่อน แล้วค่อยกลับมาโชว์หน้า
                face_hold, face_dirty = now + SAY_HOLD_MS, True
            if pump_ms and not was_on:  # นับรอบเฉพาะตอนเริ่มพ่นใหม่ สั่งต่อเวลาไม่นับซ้ำ
                bursts += 1
                t_auto = now            # คนสั่งพ่นแล้ว พ่นอัตโนมัติต้องรอพักด้วย
                w["bursts"].text("พ่นแล้ว %d รอบ" % bursts)

        # พ่นอัตโนมัติ: เฉพาะตอนแห้ง (band 0) ปั๊มว่าง ถังพอ และพ้นช่วงพักแล้ว
        if (auto_on and band == 0 and not pump_ms and tank >= TANK_MIN
                and time.ticks_diff(now, t_auto) >= auto_total):
            pump_ms, pump_t0, t_auto = AUTO_SEC * 1000, now, now
            bursts += 1
            w["bursts"].text("พ่นแล้ว %d รอบ" % bursts)
            show_note(w, "พ่นอัตโนมัติ %d วิ (แห้ง)" % AUTO_SEC, COL_MIST)
            beep("start")

        left = pump_ms - time.ticks_diff(now, pump_t0) if pump_ms else 0
        if pump_ms and (left <= 0 or tank < TANK_MIN):   # ปั๊มหมอกหยุดเองเมื่อครบเวลาหรือถังแห้ง
            pump_ms = 0
            show_note(w, "ครบเวลา หยุดพ่นเอง" if left <= 0 else "ถังหมอกแห้ง หยุดพ่นเอง", COL_MIST)
            beep("stop")
        set_led(pump, pump_ms)
        if (1 if pump_ms else 0) != last_pump:     # ปั๊มเริ่ม/หยุด: บอกเว็บทันที ไม่ต้องรอรอบถัดไป
            last_pump, send_now = 1 if pump_ms else 0, True
        sec_left = (left + 999) // 1000 if pump_ms else 0     # ปัดขึ้น: สั่ง 10 วิ เริ่มนับที่ 10 ไม่ใช่ 11
        if sec_left != shown:
            if sec_left == 0:
                face_dirty = True       # นับถอยหลังจบ จอ 8x8 ถูกล้าง ต้องวาดหน้าใหม่
            shown = matrix_countdown(sec_left, shown)
            show_pump(w, sec_left)
        # จอ 8x8 โชว์หน้าเฉพาะตอนว่าง (ไม่นับถอยหลัง ไม่เลื่อนข้อความ) วาดใหม่เมื่อช่วงเปลี่ยนเท่านั้น
        if not pump_ms and time.ticks_diff(face_hold, now) <= 0 and (face_dirty or band != face_band):
            face_dirty, face_band = False, band
            try:
                drawn = draw_face(band)
            except Exception:           # จอ 8x8 วาดไม่ได้ ไม่ควรทำให้ทั้งโปรแกรมล้ม
                drawn = False
            if not drawn and not face_warned:
                face_warned = True
                show_note(w, "rgbmatrix ไม่มีฟังก์ชันวาดจุด แก้ PIXEL_FUNCS", COL_DRY)
        tick()
        show_air(w, rh, band, tank, rh_min, rh_max, manual)
        tick()
        show_env(w, temp, temp_src, vpd, stress, auto_on)
        tick()
        gap = time.ticks_diff(now, t_send)
        if gap >= SEND_MS or (send_now and gap >= MIN_GAP_MS):   # ตามรอบ หรือมีเหตุการณ์ใหม่
            t_send, n, send_now = now, n + 1, False
            wait = max(0, auto_total - time.ticks_diff(now, t_auto)) // 1000 if auto_on else 0
            extra = {"temp_c": round(temp, 1), "vpd": round(vpd, 2), "stress": stress,
                     "auto": 1 if auto_on else 0, "auto_wait": wait, "auto_sec": AUTO_SEC,
                     "src": "VR3" if manual else "VR1", "dry": RH_DRY, "wet": RH_WET,
                     "temp_src": temp_src, "sensor_err": _temp_err,
                     "sim": "soil tank" + ("" if temp_src == "SHT40" else " temp_c")}   # sim = คีย์ที่มาจากลูกบิดจำลอง
            if not send_report(n, rh, tank, pump_ms, bursts, rh_min, extra):
                stop(w, pump, "สายหลุด")
        if not mqtt.is_connected():
            stop(w, pump, "สายหลุด")
        ui.poll()
        wait_ms(POLL_MS, BTNS)
    stop(w, pump, "เลิกฟัง พ่นไป %d รอบ" % bursts, COL_DIM)


try:
    main()
except Stop:
    pass
finally:
    try:
        mqtt.disconnect()
    except Exception:
        pass
